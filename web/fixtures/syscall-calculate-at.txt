calculateAt(dirfd, path, allowEmpty) {
  if (PATH.isAbs(path)) {
    return path;
  }
  // relative path
  var dir;
  if (dirfd === -100) {
    dir = FS.cwd();
  } else {
    var dirstream = SYSCALLS.getStreamFromFD(dirfd);
    dir = dirstream.path;
  }
  if (path.length == 0) {
    if (!allowEmpty) {
      throw new FS.ErrnoError(44);;
    }
    return dir;
  }
  return dir + '/' + path;
}
